const STORAGE_KEY = 'montfortFarewellPasses'; const LEGACY_KEY = 'farewellEpassDemo'; const PASS_LIMIT = 300; const form = document.getElementById('studentForm'); const studentNameInput = document.getElementById('studentName'); const phoneInput = document.getElementById('phoneNumber'); const previewName = document.getElementById('previewName'); const previewPhone = document.getElementById('previewPhone'); const previewCode = document.getElementById('previewCode'); const canvas = document.getElementById('qrCanvas'); const statusPill = document.getElementById('statusPill'); function normalizePhone(value) { return value.replace(/\s+/g, '').trim(); } function getStoredPasses() { try { const saved = localStorage.getItem(STORAGE_KEY); if (saved) return JSON.parse(saved); } catch (error) { console.warn('Could not parse current pass data', error); } try { const legacy = localStorage.getItem(LEGACY_KEY); if (legacy) { const parsed = JSON.parse(legacy); if (Array.isArray(parsed)) { localStorage.setItem(STORAGE_KEY, JSON.stringify(parsed)); return parsed; } } } catch (error) { console.warn('Could not parse legacy pass data', error); } return []; } function savePasses(items) { localStorage.setItem(STORAGE_KEY, JSON.stringify(items.slice(0, PASS_LIMIT))); } function generatePassCode() { const random = Math.random().toString(36).slice(2, 8).toUpperCase(); return `MWS-${new Date().getFullYear()}-${random}`; } function safeCanvasClear() { const context = canvas.getContext('2d'); if (context) { context.clearRect(0, 0, canvas.width, canvas.height); } } function renderPreview(pass) { const name = pass?.name || 'Your Name'; const phone = pass?.phone || '+91 ---------'; const id = pass?.id || '--'; previewName.textContent = name; previewPhone.textContent = phone; previewCode.textContent = id; statusPill.textContent = pass?.status || 'Pending'; statusPill.style.color = pass?.status === 'Checked In' ? '#52dbb5' : '#f6dca8'; const payload = JSON.stringify({ id: pass?.id || '', name, phone, createdAt: pass?.createdAt || new Date().toISOString(), }); safeCanvasClear(); if (pass?.id) { QRCode.toCanvas( canvas, payload, { width: 220, margin: 1, color: { dark: '#0a1626', light: '#f9f4ef', }, }, function (error) { if (error) { console.error('QR generation failed', error); } } ); } } function createPassRecord(name, phone) { const timestamp = new Date().toISOString(); return { id: generatePassCode(), name, phone, createdAt: timestamp, scannedAt: '', status: 'Pending', }; } function storePass(pass) { const list = getStoredPasses(); const updated = [pass, ...list]; savePasses(updated); } form.addEventListener('submit', function (event) { event.preventDefault(); const name = studentNameInput.value.trim(); const phone = normalizePhone(phoneInput.value); if (!name || !phone) { alert('Please enter both your name and phone number.'); return; } const pass = createPassRecord(name, phone); storePass(pass); renderPreview(pass); form.reset(); }); function copyCode() { const code = previewCode.textContent; if (!code || code === '--') return; if (navigator.clipboard && window.isSecureContext) { navigator.clipboard.writeText(code) .then(() => alert('Pass ID copied.')) .catch(() => fallbackCopy(code)); return; } fallbackCopy(code); } function fallbackCopy(value) { const helper = document.createElement('textarea'); helper.value = value; document.body.appendChild(helper); helper.select(); try { document.execCommand('copy'); alert('Pass ID copied.'); } catch (error) { alert('Copy failed. Please copy the pass ID manually: ' + value); } document.body.removeChild(helper); } document.getElementById('copyPass').addEventListener('click', copyCode); const initialPass = { id: 'MWS-2026-DEMO', name: 'Montfort Student', phone: '+91 98765 43210', createdAt: new Date().toISOString(), status: 'Pending', }; renderPreview(initialPass);
